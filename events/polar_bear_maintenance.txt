namespace = polarbear

#Give polar bear trait to polar bears born with polar bear ethnicity
polarbear.0001 = {
	hidden = yes
	trigger = {
		OR = {
			real_father = {
				culture = {
					has_cultural_pillar = heritage_polarbear
				}
			}
			mother = {
				culture = {
					has_cultural_pillar = heritage_polarbear
				}
			}
		}
	}
	immediate = {
		add_trait = polarbear_trait
		if = {
			limit = {
				NOT = {
					culture = {
						has_cultural_pillar = heritage_polarbear
					}
				}
			}
			if = {
				limit = { # if not both bears, take bear parent culture
					NAND = {
						real_father = {
							culture = {
								has_cultural_pillar = heritage_polarbear
							}
						}
						mother = {
							culture = {
								has_cultural_pillar = heritage_polarbear
							}
						}
					}
				}
				if = {
					limit = {
						real_father = {
							culture = {
								has_cultural_pillar = heritage_polarbear
							}	
						}
					}
					set_culture = real_father.culture
				}
				else = {
					set_culture = mother.culture
				}
			}
			else = {
				set_culture = mother.culture # if both parents bear take after mother
			}
		}
		add_secret = {
			type = secret_polar_bear
		}
	}
}

#Double check your polar bears on the game starting
polarbear.0002 = {
	hidden = yes
	immediate = {
		every_living_character = {
			limit = {
				culture = {
					has_cultural_pillar = heritage_polarbear
				}
				NOT = {
					has_trait = polarbear_trait
				}
			}
			add_trait = polarbear_trait
			add_secret = {
				type = secret_polar_bear
			}
		}
	}
}

#Align polar bear culture and trait for appearance
polarbear.0003 = {
	hidden = yes
	trigger = {
		OR = {
			has_trait = polarbear_trait
			culture = {
				has_cultural_pillar = heritage_polarbear
			}
		}
	}
	immediate = {
		if = {
			limit = {
				NOT = {
					culture = {
						has_cultural_pillar = heritage_polarbear
					}
				}
			}
			set_culture = culture:polarbear
		}
		add_trait = polarbear_trait
		add_secret = {
			type = secret_polar_bear
		}
	}
}

polarbear.0100 = {
	hidden = yes
	immediate = {
		set_culture = scope:new_culture
		if = {
			limit = { any_spouse = { is_playable_character = no } }
			every_spouse = {
				limit = {
					is_playable_character = no
					culture = culture:polarbear
				}
				set_culture = scope:new_culture
			}
		}
		every_close_family_member = {
			limit = {
				is_playable_character = no
			}
			custom = all_unlanded_family_members
			set_culture = scope:new_culture
		}
		every_child = {
			custom = all_children_custom
			set_culture = scope:new_culture
		}
	
		#Convert appropriate vassals, and their family
		every_vassal_or_below = {
			custom = every_polar_bear_vassal_or_below
			limit = {
				is_ai = yes
				culture = culture:polarbear
			}
			set_culture = scope:new_culture
			hidden_effect = {
				if = {
					limit = { any_spouse = { is_playable_character = no } }
					every_spouse = {
						limit = {
							is_playable_character = no
							culture = culture:polarbear
						}
						set_culture = scope:new_culture
					}
				}
				every_close_family_member = {
					limit = {
						is_playable_character = no
					}
					set_culture = scope:new_culture
				}
				every_child = {
					set_culture = scope:new_culture
				}
			}
		}
	}
}
