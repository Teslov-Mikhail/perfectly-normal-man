####
#on_game_start to check if any characters have the polarbear culture and adds the polarbear trait if they don't have it
on_game_start_after_lobby = {
    on_actions = {
        polarbear_on_action 
    }
}

polarbear_on_action = {
	effect = {
        every_player = {
			trigger_event = {
				id = polarbear.0002
			}
        }
    }
}

# fix for issue #3
on_character_created = {
    on_actions = {	
        polarbear_on_character_created
    }
}

polarbear_on_character_created = {
	effect = {
		if = {
			limit = {
				culture = culture:polarbear
				NOT = {
					has_trait = polarbear_trait
				}
			}
			add_trait = polarbear_trait
			add_secret = {
				type = secret_polar_bear
			}
		}
	}
}
